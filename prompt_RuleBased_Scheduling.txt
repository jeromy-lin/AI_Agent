df_wo = sheets_df['生產工單'].copy()

# 2.1 日期與數值轉型
df_wo['開單日期'] = pd.to_datetime(df_wo['開單日期'], errors='coerce')
df_wo['交期'] = pd.to_datetime(df_wo['交期'], errors='coerce')

df_wo['計畫數量'] = pd.to_numeric(df_wo['計畫數量'], errors='coerce').fillna(0)
df_routing['標準換線工時_hr'] = pd.to_numeric(df_routing['標準換線工時_hr'], errors='coerce').fillna(0.0)
df_routing['標準加工工時_hr每100件'] = pd.to_numeric(df_routing['標準加工工時_hr每100件'], errors='coerce').fillna(0.0)
df_machine['每日可用工時'] = pd.to_numeric(df_machine['每日可用工時'], errors='coerce').fillna(0.0)

print("未完工狀態明細：")
print(df_uncompleted['狀態'].value_counts().to_string())

# ==============================================================================

valid_dates = df_uncompleted['開單日期'].dropna()
if len(valid_dates) == 0:
    PLANNING_DATE = pd.Timestamp.now().floor('D')

    PLANNING_DATE = valid_dates.min()

print(f"\n排程基準日 (PLANNING_DATE)：{PLANNING_DATE.strftime('%Y-%m-%d')}")

# 計算距離交期天數 (交期缺失則填入 999)
df_uncompleted['距離交期天數'] = (df_uncompleted['交期'] - PLANNING_DATE).dt.days
df_uncompleted['距離交期天數'] = df_uncompleted['距離交期天數'].fillna(999).astype(int)

# ==============================================================================
# STEP 4: CNC 標準工時計算與路由對應
# ==============================================================================
# 篩選 CNC 工序
df_cnc_routing = df_routing[df_routing['需求工作中心'].astype(str).str.upper() == 'CNC'].copy()

cnc_sku_summary = df_cnc_routing.groupby('SKU').agg(
    CNC工序數=('工序序號', 'count'),
    CNC標準換線工時_hr=('標準換線工時_hr', 'sum'),
    CNC標準加工工時_hr每100件=('標準加工工時_hr每100件', 'sum')
).reset_index()

# 串接至未完工工單
df_uncompleted = pd.merge(df_uncompleted, cnc_sku_summary, on='SKU', how='left')

# 判斷是否有 CNC 路由與缺失值補0
df_uncompleted['具有CNC路由'] = df_uncompleted['CNC工序數'].notnull() & (df_uncompleted['CNC工序數'] > 0)
df_uncompleted['CNC工序數'] = df_uncompleted['CNC工序數'].fillna(0).astype(int)
df_uncompleted['CNC標準換線工時_hr'] = df_uncompleted['CNC標準換線工時_hr'].fillna(0.0)
df_uncompleted['CNC標準加工工時_hr每100件'] = df_uncompleted['CNC標準加工工時_hr每100件'].fillna(0.0)

# 計算工時
df_uncompleted['CNC加工工時_hr'] = (df_uncompleted['CNC標準加工工時_hr每100件'] * df_uncompleted['計畫數量'] / 100.0).round(2)
df_uncompleted['CNC排程總工時_hr'] = (df_uncompleted['CNC標準換線工時_hr'] + df_uncompleted['CNC加工工時_hr']).round(2)

# ==============================================================================
# STEP 5: Rule-Based Priority Score 評分機制
# ==============================================================================
def calc_urgent_score(val):
    return 30 if str(val).strip() == '是' else 0

def calc_due_score(days):
    if days <= 1:
        return 30
    elif days <= 2:
        return 25
    elif days <= 4:
        return 15
    elif days <= 7:
        return 5
    else:
        return 0

def calc_cust_score(val):
    v = str(val).strip().upper()
    if v in ['VIP', 'VIP客戶']:
        return 20
    elif v in ['重要', '重要客戶']:
        return 10
    else:
        return 0

def calc_mat_score(val):
    return 15 if str(val).strip() == '是' else -35

def calc_prio_score(val):
    v = str(val).strip()
    if v == '高':
        return 10
    elif v == '中':
        return 5
    else:
        return 0

def calc_changeover_score(row):
    if not row['具有CNC路由']:
        return 0
    hrs = row['CNC標準換線工時_hr']
    if hrs <= 1.5:
        return 10
    elif hrs >= 3.5:
        return -5
    else:
        return 0

def calc_qual_score(val):
    v = str(val).strip()
    if v in ['高風險', '高']:
        return -15
    elif v in ['中風險', '中']:
        return -5
    else:
        return 0

df_uncompleted['急單分數'] = df_uncompleted['急單'].apply(calc_urgent_score)
df_uncompleted['交期分數'] = df_uncompleted['距離交期天數'].apply(calc_due_score)
df_uncompleted['客戶分數'] = df_uncompleted['客戶等級'].apply(calc_cust_score)
df_uncompleted['材料分數'] = df_uncompleted['材料齊套'].apply(calc_mat_score)
df_uncompleted['工單優先等級分數'] = df_uncompleted['優先等級'].apply(calc_prio_score)
df_uncompleted['換線分數'] = df_uncompleted.apply(calc_changeover_score, axis=1)
df_uncompleted['品質分數'] = df_uncompleted['品質風險'].apply(calc_qual_score)

df_uncompleted['排程分數'] = (
    df_uncompleted['急單分數'] +
    df_uncompleted['交期分數'] +
    df_uncompleted['客戶分數'] +
    df_uncompleted['材料分數'] +
    df_uncompleted['工單優先等級分數'] +
    df_uncompleted['換線分數'] +
    df_uncompleted['品質分數']
)

# ==============================================================================
# STEP 6: 可立即排產篩選與規則說明
# ==============================================================================
def check_ready(row):
    cond1 = str(row['材料齊套']).strip() == '是'
    cond2 = str(row['品質風險']).strip() not in ['高風險', '高']
    cond3 = bool(row['具有CNC路由'])
    cond4 = pd.notnull(row['CNC排程總工時_hr']) and row['CNC排程總工時_hr'] > 0
    return '是' if (cond1 and cond2 and cond3 and cond4) else '否'

df_uncompleted['可立即排產'] = df_uncompleted.apply(check_ready, axis=1)

def build_rule_desc(row):
    parts = []
    if row['急單分數'] > 0:
        parts.append(f"急單 +{row['急單分數']}")
    if row['交期分數'] > 0:
        parts.append(f"交期壓力 +{row['交期分數']}")
    if row['客戶分數'] > 0:
        parts.append(f"客戶等級 +{row['客戶分數']}")
    if row['材料分數'] != 0:
        parts.append(f"材料狀態 {row['材料分數']:+d}")
    if row['工單優先等級分數'] > 0:
        parts.append(f"優先等級 +{row['工單優先等級分數']}")
    if row['換線分數'] != 0:
        parts.append(f"換線工時 {row['換線分數']:+d}")
    if row['品質分數'] != 0:
        parts.append(f"品質風險 {row['品質分數']:+d}")
    return "；".join(parts) if parts else "基本評分"

df_uncompleted['規則說明'] = df_uncompleted.apply(build_rule_desc, axis=1)

# 子集分類
df_ready = df_uncompleted[df_uncompleted['可立即排產'] == '是'].copy()
df_pending_mat = df_uncompleted[df_uncompleted['材料齊套'].astype(str).str.strip() == '否'].copy()
df_high_risk = df_uncompleted[df_uncompleted['品質風險'].astype(str).str.strip().isin(['高風險', '高'])].copy()

# ==============================================================================
# STEP 7: 未完工工單多重排序與排程順位
# ==============================================================================
df_sorted = df_uncompleted.sort_values(
    by=['可立即排產', '排程分數', '距離交期天數', 'CNC標準換線工時_hr', '開單日期'],
    ascending=[False, False, True, True, True]
).reset_index(drop=True)

df_sorted['排程順位'] = range(1, len(df_sorted) + 1)

# ==============================================================================
# STEP 8: CNC 負載平衡動態派工 (Rule-Based)
# ==============================================================================
cnc_machines = df_machine[df_machine['工作中心'].astype(str).str.upper() == 'CNC'].copy()
if (cnc_machines['狀態'].astype(str).str.strip() == '可用').any():
    cnc_machines = cnc_machines[cnc_machines['狀態'].astype(str).str.strip() == '可用'].copy()

machine_ids = sorted(cnc_machines['機台ID'].tolist())

# 可排產工單依 Rule-Based 順序派工
df_ready_sorted = df_sorted[df_sorted['可立即排產'] == '是'].copy().reset_index(drop=True)
df_ready_sorted['RuleBased順位'] = range(1, len(df_ready_sorted) + 1)

machine_load_rb = {m: 0.0 for m in machine_ids}
rb_assignments = []

for idx, row in df_ready_sorted.iterrows():
    min_m = min(machine_load_rb, key=machine_load_rb.get)
    start_t = machine_load_rb[min_m]
    dur = row['CNC排程總工時_hr']
    end_t = start_t + dur
    machine_load_rb[min_m] = end_t
    
    rb_assignments.append({
        '工單ID': row['工單ID'],
        'RuleBased_機台': min_m,
        'RuleBased_開始工時': round(start_t, 2),
        'RuleBased_排程總工時': round(dur, 2),
        'RuleBased_完成工時': round(end_t, 2)
    })

df_rb_dispatch = pd.DataFrame(rb_assignments)
df_ready_sorted = pd.merge(df_ready_sorted, df_rb_dispatch, on='工單ID', how='left')

# ==============================================================================
# STEP 9: FIFO Baseline 派工基準與比較
# ==============================================================================
df_fifo = df_ready_sorted.copy()
df_fifo = df_fifo.sort_values(by=['開單日期', '工單ID'], ascending=[True, True]).reset_index(drop=True)
df_fifo['FIFO順位'] = range(1, len(df_fifo) + 1)

machine_load_fifo = {m: 0.0 for m in machine_ids}
fifo_assignments = []

for idx, row in df_fifo.iterrows():
    min_m = min(machine_load_fifo, key=machine_load_fifo.get)
    start_t = machine_load_fifo[min_m]
    dur = row['CNC排程總工時_hr']
    end_t = start_t + dur
    machine_load_fifo[min_m] = end_t
    
    fifo_assignments.append({
        '工單ID': row['工單ID'],
        'FIFO_機台': min_m,
        'FIFO_開始工時': round(start_t, 2),
        'FIFO_排程總工時': round(dur, 2),
        'FIFO_完成工時': round(end_t, 2)
    })

df_fifo_dispatch = pd.DataFrame(fifo_assignments)
df_fifo = pd.merge(df_fifo, df_fifo_dispatch, on='工單ID', how='left')

# 整合 FIFO 與 Rule-Based 工單順位比較
df_comp_wo = pd.merge(
    df_ready_sorted[['RuleBased順位', '工單ID', 'SKU', '急單', '客戶等級', '開單日期', '交期', '距離交期天數', '排程分數', 'RuleBased_機台', 'RuleBased_開始工時', 'RuleBased_完成工時']],
    df_fifo[['工單ID', 'FIFO順位', 'FIFO_機台', 'FIFO_開始工時', 'FIFO_完成工時']],
    on='工單ID',
    how='left'
)
df_comp_wo['順位提前'] = df_comp_wo['FIFO順位'] - df_comp_wo['RuleBased順位']

# ==============================================================================
# STEP 10: 效益指標分析與度量
# ==============================================================================
def safe_improvement_rate(fifo_val, rb_val):
    if fifo_val == 0 or pd.isna(fifo_val):
        return 0.0
    return round((fifo_val - rb_val) / fifo_val * 100.0, 2)

urgent_wo = df_comp_wo[df_comp_wo['急單'].astype(str).str.strip() == '是']
urgent_count = len(urgent_wo)
urgent_fifo_avg = urgent_wo['FIFO順位'].mean() if urgent_count > 0 else 0.0
urgent_rb_avg = urgent_wo['RuleBased順位'].mean() if urgent_count > 0 else 0.0

vip_wo = df_comp_wo[df_comp_wo['客戶等級'].astype(str).str.strip().str.upper().isin(['VIP', 'VIP客戶'])]
vip_count = len(vip_wo)
vip_fifo_avg = vip_wo['FIFO順位'].mean() if vip_count > 0 else 0.0
vip_rb_avg = vip_wo['RuleBased順位'].mean() if vip_count > 0 else 0.0

makespan_fifo = max(machine_load_fifo.values()) if machine_load_fifo else 0.0
makespan_rb = max(machine_load_rb.values()) if machine_load_rb else 0.0

# 機台利用率計算
df_mach_util = pd.DataFrame({'機台ID': machine_ids})
df_mach_util['FIFO_累積工時_hr'] = df_mach_util['機台ID'].map(machine_load_fifo).round(2)
df_mach_util['RuleBased_累積工時_hr'] = df_mach_util['機台ID'].map(machine_load_rb).round(2)

df_mach_util['FIFO_機台利用率_%'] = (df_mach_util['FIFO_累積工時_hr'] / makespan_fifo * 100.0).round(1) if makespan_fifo > 0 else 0.0
df_mach_util['RuleBased_機台利用率_%'] = (df_mach_util['RuleBased_累積工時_hr'] / makespan_rb * 100.0).round(1) if makespan_rb > 0 else 0.0
df_mach_util['利用率百分點變化'] = (df_mach_util['RuleBased_機台利用率_%'] - df_mach_util['FIFO_機台利用率_%']).round(1)

avg_util_fifo = df_mach_util['FIFO_機台利用率_%'].mean() if len(df_mach_util) > 0 else 0.0
avg_util_rb = df_mach_util['RuleBased_機台利用率_%'].mean() if len(df_mach_util) > 0 else 0.0
util_pct_diff = round(avg_util_rb - avg_util_fifo, 1)

# 效益彙整表
ready_count = len(df_ready)
ready_ratio = round(ready_count / uncompleted_wo_count * 100.0, 1) if uncompleted_wo_count > 0 else 0.0
pending_mat_count = len(df_pending_mat)
pending_mat_ratio = round(pending_mat_count / uncompleted_wo_count * 100.0, 1) if uncompleted_wo_count > 0 else 0.0
high_risk_count = len(df_high_risk)
high_risk_ratio = round(high_risk_count / uncompleted_wo_count * 100.0, 1) if uncompleted_wo_count > 0 else 0.0

df_efficiency = pd.DataFrame([
    {'指標項目': '未完工總工單數', 'FIFO Baseline': str(uncompleted_wo_count), 'Rule-Based': str(uncompleted_wo_count), '差異 / 改善率': '-'},
    {'指標項目': '可立即排產工單數 (比率)', 'FIFO Baseline': f"{ready_count} ({ready_ratio}%)", 'Rule-Based': f"{ready_count} ({ready_ratio}%)", '差異 / 改善率': '-'},
    {'指標項目': '待料工單數 (比率)', 'FIFO Baseline': f"{pending_mat_count} ({pending_mat_ratio}%)", 'Rule-Based': f"{pending_mat_count} ({pending_mat_ratio}%)", '差異 / 改善率': '-'},
    {'指標項目': '高品質風險工單數 (比率)', 'FIFO Baseline': f"{high_risk_count} ({high_risk_ratio}%)", 'Rule-Based': f"{high_risk_count} ({high_risk_ratio}%)", '差異 / 改善率': '-'},
    {'指標項目': '急單平均派工順位', 'FIFO Baseline': round(urgent_fifo_avg, 2), 'Rule-Based': round(urgent_rb_avg, 2), '差異 / 改善率': f"{safe_improvement_rate(urgent_fifo_avg, urgent_rb_avg)}%"},
    {'指標項目': 'VIP平均派工順位', 'FIFO Baseline': round(vip_fifo_avg, 2), 'Rule-Based': round(vip_rb_avg, 2), '差異 / 改善率': f"{safe_improvement_rate(vip_fifo_avg, vip_rb_avg)}%"},
    {'指標項目': '總排程工期 Makespan (hr)', 'FIFO Baseline': round(makespan_fifo, 2), 'Rule-Based': round(makespan_rb, 2), '差異 / 改善率': f"{safe_improvement_rate(makespan_fifo, makespan_rb)}%"},
    {'指標項目': '平均 CNC 利用率 (%)', 'FIFO Baseline': f"{round(avg_util_fifo, 1)}%", 'Rule-Based': f"{round(avg_util_rb, 1)}%", '差異 / 改善率': f"{util_pct_diff:+.1f} 百分點"}
])

# STEP 11: 輸出 11 張 Pandas Styler 表格
# ==============================================================================
def display_styled_table(df, title):
    print(f"\n" + "="*80)
    print(f"  {title}")
    print("="*80)
    
    df_show = df.copy()
    
    # 格式化日期與數值欄位
    for col in df_show.columns:
        if pd.api.types.is_datetime64_any_dtype(df_show[col]):
            df_show[col] = df_show[col].dt.strftime('%Y-%m-%d')
    
    df_clean = df_show.fillna("-")
    
    styler = df_clean.style.set_table_styles([
        {'selector': 'th', 'props': [('background-color', '#2B4C7E'), ('color', '#FFFFFF'), ('font-weight', 'bold'), ('text-align', 'center'), ('padding', '8px'), ('font-size', '13px')]},
        {'selector': 'td', 'props': [('text-align', 'center'), ('padding', '6px'), ('font-style', 'normal'), ('font-size', '12px')]},
        {'selector': 'tr:nth-child(even)', 'props': [('background-color', '#F4F7FA')]},
        {'selector': 'tr:nth-child(odd)', 'props': [('background-color', '#FFFFFF')]}
    ]).hide(axis='index')
